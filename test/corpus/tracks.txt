instrument bass {
  gain 1.8
  osc saw(55) as osc1
  osc saw(55, pitch=-12) as osc2
  osc1 > mix
  osc2 > mix
  mix > gain(0.333) as prescale
  prescale > ladder(50, 0.7, ea=0.005, edepth=1500) as filter
  filter > adsr(0.005, 0.2, 0.8, 0.15) as amp
  amp > out
}
